// Key API patterns:
// - Render: fn render(&mut self, _window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement
// - Return div().into_any_element() when returning from Render
// - Colors: rgb(0x...) returns Rgba, use directly (no Hsla)
// - Mouse: on_mouse_down(MouseButton::Left/Right, cx.listener(|this, event: &EventType, _w, cx| {}))
// - Keyboard: on_key_down(cx.listener(|this, event: &KeyDownEvent, _, cx| {}))
// - Button: button::Button::new("id").label("text").on_click(cx.listener(...))
// - Image: img(ImageSource::Render(arc)).w(px(n)).h(px(n))
// - Context: cx.listener takes 4 args: |this, event, window, cx|
// - Clipboard: use cx.write_to_clipboard(...) — or just skip and use platform clipboard crate
// - Root::new(view, window, cx) wraps window content
// - gpui_component::init(cx) called early in app.run
// - gpui_platform::application().with_assets(gpui_component_assets::Assets)
// - cx.spawn(async move |cx| { cx.open_window(...) }).detach()
// - Modal dialogs: div().absolute().inset_0().bg(rgba(0,0,0,0.4)).flex().items_center().justify_center()
// - Text: cx.listener for text change is tricky — use button:Button to trigger dialog
// - Clipboard: Actually in GPUI this is complex, skip clipboard API for now
// - Open in Explorer: std::process::Command::new("explorer").arg("/select,").arg(&path).spawn()
